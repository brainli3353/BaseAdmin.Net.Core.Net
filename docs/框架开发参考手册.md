## 1.1 两阶段登录(账号密码初验 → OTP 挑战兑换)

### 结构树

```
Business/Services/Auth/
├── AuthService.cs
│   ├── LoginAsync(LoginInput) → LoginOutput        # 阶段一:密码初验
│   │   └── 错误计数 → 查用户 → 状态校验 → PBKDF2 校验 → CreateLoginOtpChallenge
│   ├── LoginOtpAsync(LoginOtpInput) → LoginOutput   # 阶段二:OTP 兑换
│   │   └── 取挑战 → FailTimes 检查 → OtpService.Validate → 首次绑定/更新LastUsedTimeStep → CreateToken
│   ├── CreateToken(SysUser, LoginMode) → LoginOutput
│   │   └── 单设备踢人 → JWT 签发 → 缓存 TokenVersion → 登录事件
│   └── LogoutAsync()                                 # 登出:Token 写黑名单
├── Dto/
│   ├── LoginInput.cs(Account/Password/TenantId/LoginMode)
│   ├── LoginOtpInput.cs(ChallengeId/OtpCode)
│   └── LoginOutput.cs(Token/RefreshToken/ChallengeId/BindRequired/QrCodeBase64/ManualSecret)
├── LoginOtpChallenge.cs                              # Redis 挑战载荷 record
└── (框架层)Core/Security/OtpService.cs + CryptogramHelper.cs
```

### 核心代码参考

```csharp
// 阶段一:密码初验 → 只发挑战,不发 Token(核心安全设计)
public async Task<LoginOutput> LoginAsync(LoginInput input)
{
    // 1. 密码错误次数 Redis 滑动窗口(30 分钟 TTL,超 5 次拒绝)
    var key = $"{CacheConst.KeyPasswordErrorTimes}{input.Account}";
    var errorTimes = _cache.Get<int>(key);
    if (errorTimes > 5) throw new AppException("密码错误次数过多,请稍后再试");

    // 2. 查用户(Account 匹配)+ 状态校验
    var user = await _userRep.AsQueryable().FirstAsync(u => u.Account == input.Account);
    if (user == null || user.Status != 0) throw new AppException("账号或密码错误");

    // 3. PBKDF2 校验(失败 → 计数+1 写回)
    if (!Pbkdf2Helper.VerifyPassword(input.Password, user.Password))
    {
        _cache.Set(key, errorTimes + 1, TimeSpan.FromMinutes(30));
        throw new AppException("账号或密码错误");
    }
    _cache.Remove(key); // 成功清计数

    // 4. 生成挑战(300 秒,一次性)——密码对了也不发 Token
    var challengeId = YitIdHelper.NextId().ToString();
    _cache.Set($"{CacheConst.KeyLoginOtpChallenge}{challengeId}",
        new LoginOtpChallengeCache(user.Id, /* SetupSecret 首次绑定用 */ null, 0),
        TimeSpan.FromSeconds(300));
    return new LoginOutput { ChallengeId = challengeId, BindRequired = /* 未绑定 */ };
}
```

```csharp
// 阶段二:OTP 校验兑换(防重放:同一时间步不可复用)
public async Task<LoginOutput> LoginOtpAsync(LoginOtpInput input)
{
    var challenge = _cache.Get<LoginOtpChallengeCache>($"{CacheConst.KeyLoginOtpChallenge}{input.ChallengeId}")
        ?? throw new AppException("验证已过期,请重新登录");
    if (challenge.FailTimes >= 5) throw new AppException("验证码错误次数过多");

    var userOtp = await _userOtpRep.GetFirstAsync(u => u.UserId == challenge.UserId && u.Enabled);
    var secret = userOtp == null ? challenge.SetupSecret : CryptogramHelper.Decrypt(userOtp.SecretCipher);

    var step = OtpService.Validate(input.OtpCode, secret, userOtp?.LastUsedTimeStep); // 前后各 1 窗口 + 防重放
    if (step == null)
    {
        challenge.FailTimes++; _cache.Set(key, challenge, TimeSpan.FromSeconds(300));
        throw new AppException("验证码错误");
    }
    if (userOtp == null) { /* 首次:绑定持久化(SecretCipher 加密) */ }
    else { userOtp.LastUsedTimeStep = step; await _userOtpRep.UpdateAsync(userOtp); }
    _cache.Remove(key); // 一次性挑战即焚
    return await CreateToken(await _userRep.GetByIdAsync(challenge.UserId));
}
```

```csharp
// Token 校验链:自动续期 → 黑名单 → TokenVersion → 账号状态(顺序重要)
protected override Task HandleRequirementAsync(AuthorizationHandlerContext context, JwtAuthRequirement requirement)
{
    var user = context.User;
    // 1. 黑名单(登出时写入)
    var token = context.Resource as string; // 原始 Token
    if (_cache.ExistKey($"{CacheConst.KeyTokenBlacklist}:{token}")) { context.Fail(); return Task.CompletedTask; }

    // 2. TokenVersion(改密/踢人时 +1,旧 Token 全失效——无状态实现的核心)
    var tv1 = user.FindFirst(JwtClaimNames.TokenVersion)?.Value;
    var tv2 = _cache.Get<string>($"{CacheConst.KeyUserToken}{userId}");
    if (tv2 == null) { tv2 = /* DB 回退查用户 */; _cache.Set(...); }
    if (tv1 != tv2) { context.Fail(); return Task.CompletedTask; }

    // 3. 账号状态(停用拒绝)
    context.Succeed(requirement);
    return Task.CompletedTask;
}
```

### Furion 依赖替换点

| baseadmin(Furion) | 原生 |
|---|---|
| `JWTEncryption.AutoRefreshToken`(自动续期) | 原生 `OnTokenValidated` 事件里重签 Token + 响应头回写 |
| `AppAuthorizeHandler` | `AuthorizationHandler<IAuthorizationRequirement>` |

### 开发注意

- **TokenVersion 是"改密即踢全端"的关键**:改密码/重置/角色变更时 `TokenVersion++` + 删缓存 + SignalR 强踢(三动作闭环)
- 缓存缺失必须 DB 回退(防重启后全员登出)
- 黑名单 TTL = Token 剩余有效期(过期自然失效)

```csharp
{
    private readonly SysTenantService _tenantService; // 构造注入,替代 baseadmin 构造体内 App.GetRequiredService

    public SqlSugarRepository(SysTenantService tenantService)
    {
        _tenantService = tenantService;
        var iTenant = SqlSugarSetup.ITenant;
        base.Context = iTenant.GetConnectionScope(SqlSugarConst.MainConfigId);

        // 若实体贴有多库特性,则返回指定库连接
        if (typeof(T).IsDefined(typeof(TenantAttribute), false))
        { base.Context = iTenant.GetConnectionScopeWithAttr<T>(); return; }
        // 若实体贴有日志表特性,则返回日志库连接
        if (typeof(T).IsDefined(typeof(LogTableAttribute), false))
        {
            if (iTenant.IsAnyConnection(SqlSugarConst.LogConfigId))
                base.Context = iTenant.GetConnectionScope(SqlSugarConst.LogConfigId);
            return;
        }
        // 若实体贴有系统表特性,则返回默认库连接
        if (typeof(T).IsDefined(typeof(SysTableAttribute), false)) return;

        // 未贴任何表特性:按当前请求租户 Claim 切库(IHttpContextAccessor 取,替代 App.User/App.HttpContext)
        var tenantId = CurrentTenantId(); // IHttpContextAccessor.HttpContext.User.FindFirst(TenantId)?.Value
        if (string.IsNullOrWhiteSpace(tenantId) || tenantId == SqlSugarConst.MainConfigId) return;
        var tenantDb = _tenantService.GetTenantDbConnectionScope(long.Parse(tenantId)); // 懒加载,见 3.10
        if (tenantDb != null) base.Context = tenantDb;
    }

    // ③ 乐观锁:SET Ver=Ver+1 WHERE Ver=@old(Expression 树构建,int/int? 均支持)
    public virtual async Task<bool> UpdateWithVerAsync(T input, string verFieldName = "Ver")
    {
        var verProperty = typeof(T).GetProperty(verFieldName);
        if (verProperty == null) // 无 Ver 字段 → 普通更新
            return await base.AsUpdateable(input).ExecuteCommandAsync() > 0;

        var originalVer = verProperty.GetValue(input);
        if (originalVer == null && verProperty.PropertyType != typeof(int))
            throw new ArgumentException($"执行并发安全更新时 {verFieldName} 不能为空。");
        if (originalVer == null) originalVer = 0;

        // Expression 树:左=it.Ver,右=(it.Ver+1),where=it.Ver==originalVer(详见 baseadmin 实现,此处从略)
        int updateResult = await base.AsUpdateable(input)
            .PublicSetColumns(BuildVerLeft<T>(verProperty), BuildVerRight<T>(verProperty))
            .Where(BuildVerWhere<T>(verProperty, originalVer))
            .ExecuteCommandAsync();
        return updateResult > 0; // false = 并发冲突或记录不存在
    }

    // ④ 分表:AsInsertable(input).SplitTable().ExecuteCommandAsync() / AsQueryable().SplitTable()……
    //    实体贴 [SplitTable] 特性,baseadmin 提供 SplitTableInsert/Update/Delete/GetFirst/IsAny/GetList 全套
}
```

```csharp
// ① 事务驱动器:自定义特性 + ActionFilter(替代 Furion [UnitOfWork] AOP;钩子体框架无关保留)
[AttributeUsage(AttributeTargets.Method)]
public class UnitOfWorkAttribute : Attribute { }

public class UnitOfWorkFilter : ActionFilterAttribute
{
    private readonly SqlSugarUnitOfWork _unitOfWork;
    public UnitOfWorkFilter(SqlSugarUnitOfWork unitOfWork) => _unitOfWork = unitOfWork;

    public override async Task OnActionExecutionAsync(ActionExecutingContext context, ActionExecutionDelegate next)
    {
        if (context.ActionDescriptor.EndpointMetadata.GetMetadata<UnitOfWorkAttribute>() == null)
        {
            await next();
            return;
        }

        _unitOfWork.BeginTransaction();
        try
        {
            var executed = await next();
            if (executed.Exception != null) _unitOfWork.RollbackTransaction();
            else _unitOfWork.CommitTransaction();
        }
        finally
        {
            _unitOfWork.OnCompleted();
        }
    }
}

public sealed class SqlSugarUnitOfWork
{
    private readonly ISqlSugarClient _sqlSugarClient;
    public SqlSugarUnitOfWork(ISqlSugarClient sqlSugarClient) => _sqlSugarClient = sqlSugarClient;

    public void BeginTransaction() => _sqlSugarClient.AsTenant().BeginTran(); // 对全部已注册连接开启事务(跨库)
    public void CommitTransaction() => _sqlSugarClient.AsTenant().CommitTran();
    public void RollbackTransaction() => _sqlSugarClient.AsTenant().RollbackTran();
    public void OnCompleted() { /* 事务对象释放;单例 SqlSugarScope 场景注意勿 Dispose 全局实例 */ }
}
```

```csharp
// 二级缓存适配器直接持 IConnectionMultiplexer 用 StackExchange.Redis 同步 API(禁 .Result/.Wait() 约定下的唯一正解)
public class SqlSugarCache : SqlSugar.ICacheService
{
    private const string Prefix = "sys_sqlSugar:";
    private readonly IDatabase _db; // ConnectionMultiplexer.GetDatabase(),与业务异步缓存同一连接实例
    public SqlSugarCache(IConnectionMultiplexer muxer) => _db = muxer.GetDatabase();

    public void Add<V>(string key, V value)
        => _db.StringSet(Prefix + key, Serialize(value));
    public void Add<V>(string key, V value, int cacheDurationInSeconds)
        => _db.StringSet(Prefix + key, Serialize(value), TimeSpan.FromSeconds(cacheDurationInSeconds));
    public bool ContainsKey<V>(string key)
        => _db.KeyExists(Prefix + key);
    public V Get<V>(string key)
    {
        var raw = _db.StringGet(Prefix + key);
        return raw ? Deserialize<V>(raw.ToString()) : default;
    }
    public IEnumerable<string> GetAllKey<V>()
        => _db.Keys(Prefix + "*").Select(k => k.ToString().Replace(Prefix, "")).ToList();
    public V GetOrCreate<V>(string key, Func<V> create, int cacheDurationInSeconds = int.MaxValue)
    {
        var raw = _db.StringGet(Prefix + key);
        if (raw) return Deserialize<V>(raw.ToString());
        var value = create();
        _db.StringSet(Prefix + key, Serialize(value),
            cacheDurationInSeconds == int.MaxValue ? (TimeSpan?)null : TimeSpan.FromSeconds(cacheDurationInSeconds));
        return value;
    }
    public void Remove<V>(string key)
        => _db.KeyDelete(key); // SqlSugar 调用 Remove 时 key 中已含前缀,勿重复加
}
// 注册:ConfigureExternalServices.DataInfoCacheService = new SqlSugarCache(muxer)(SetDbConfig 内,见 3.1)
```

```csharp
// ISqlSugarClient 单例经构造注入(替代 App.GetRequiredService);租户信息来自 ICacheService 缓存(替代 Furion 服务定位)
public SqlSugarScopeProvider GetTenantDbConnectionScope(long tenantId)
{
    var iTenant = _db.AsTenant();
    if (iTenant.IsAnyConnection(tenantId.ToString()))
        return iTenant.GetConnectionScope(tenantId.ToString());

    lock (iTenant)
    {
        var tenant = _cache.Get<List<SysTenant>>(CacheConst.KeyTenant)?.FirstOrDefault(u => u.Id == tenantId);
        if (tenant == null || tenant.TenantType == TenantTypeEnum.Id) return null; // Id 型租户无独立库

        var tenantConnConfig = new ConnectionConfig
        {
            ConfigId = tenant.Id.ToString(),
            DbType = tenant.DbType,
            IsAutoCloseConnection = true,
            ConnectionString = CryptogramHelper.SM2Decrypt(tenant.Connection) // 连接串加密存储,取用时解密
        };
        iTenant.AddConnection(tenantConnConfig);

        var scope = iTenant.GetConnectionScope(tenantId.ToString());
        SqlSugarSetup.SetDbConfig(tenantConnConfig); // 新连接补外部服务/MoreSettings
        SqlSugarSetup.SetDbAop(scope, enableConsoleSql: false); // ★ 新连接必须重挂 AOP(审计/过滤器不继承)
        return scope;
    }
}
```

```csharp
public async Task<string> NextSeqNo(string type, bool isGlobal = false)
{
    var tenantId = isGlobal ? SqlSugarConst.DefaultTenantId : _userManager.TenantId;
    using var lock = _cache.BeginCacheLock($"{CacheConst.KeySerialLock}:{tenantId}:{type}", 1000)
        ?? throw new AppException("并发获取流水号锁失败");
    var seq = await _serialRep.AsQueryable().IgnoreTenant()
        .FirstAsync(u => u.Type == type && u.Status == StatusEnum.Enable && u.TenantId == tenantId)
        ?? throw new AppException("流水号配置不存在或未启用");
    TryRestSeqNo(seq, DateTime.Now);
    seq.Seq++;
    if (seq.Seq > seq.Max) throw new AppException("流水号超出最大值");
    await _serialRep.UpdateAsync(u => new { Expy = seq.Expy, Seq = seq.Seq }, u => u.Id == seq.Id);
    return FormatSeqNo(seq, DateTime.Now);

// 插槽:委托字典(不可序列化→进程内 ConcurrentDictionary 缓存,非 Redis)
private string FormatSeqNo(SysSerial seq, DateTime nowDate)
{
    // 默认插槽:{SEQ}(按 Max 位数补零)/{yyyy}{yy}{MM}{dd}{HH}{mm}{ss}/{TenantId}{UserId}{OrgId}{OrgType}
    // 业务扩展:实现 ISerialSlotProvider,反射扫描合并(进程内缓存)
    var replacements = GetSlotMap(seq.Seq, seq.Max, nowDate);
    return Regex.Replace(seq.Formater, @"\{.*?\}", m =>
        replacements.TryGetValue(m.Value, out var f) ? f() : throw new AppException($"无效插槽: {m.Value}"));
}
```

```csharp
public class FileLoggerProvider : ILoggerProvider
{
    private readonly string _basePath;              // 配置 Logging:File:Path,默认 "logs"
    private readonly string _fileNameRule;          // 例 "log-{0:yyyy-MM-dd}-{1}" → log-2026-09-09-Information
    private readonly object _lock = new();
    private readonly Dictionary<string, FileStream> _streams = new(); // 文件名→流(按天缓存)

    public FileLoggerProvider(IConfiguration config)
    {
        _basePath = config["Logging:File:Path"] ?? "logs";
        _fileNameRule = config["Logging:File:FileNameRule"] ?? "log-{0:yyyy-MM-dd}-{1}";
        Directory.CreateDirectory(_basePath);
    }

    public ILogger CreateLogger(string categoryName) => new FileLogger(this, categoryName);

    // 写一行:开流(当日文件)→ 追加;开流/写失败 → 切换 "{name}-oops{ext}" 备用文件重试一次
    internal void Write(LogLevel level, string categoryName, string message, Exception? ex)
    {
        string path = Path.Combine(_basePath, string.Format(_fileNameRule, DateTime.Now, level.ToString()));
        try
        {
            lock (_lock)
            {
                if (!_streams.TryGetValue(path, out var fs))
                {
                    fs = new FileStream(path, FileMode.Append, FileAccess.Write, FileShare.Read);
                    _streams[path] = fs;
                }
                fs.Write(_utf8.GetBytes($"{DateTime.Now:yyyy-MM-dd HH:mm:ss.fff} [{level}] {categoryName}: {message}{Environment.NewLine}"));
                fs.Flush();
            }
        }
        catch
        {
            // 备用文件:去掉扩展名 + "-oops" + 扩展名(对齐 baseadmin HandleWriteError)
            var oops = Path.Combine(_basePath, Path.GetFileNameWithoutExtension(path) + "-oops" + Path.GetExtension(path));
            File.AppendAllText(oops, $"{DateTime.Now:O} [{level}] {categoryName}: {message}{Environment.NewLine}");
        }
    }
    // 每级别一条通道 = 三个 AddProvider(WriteFilter 按 LogLevel>=X 过滤),或单一 Provider 内部按级别分文件
}
```

```csharp
        _config = config;

    protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken ct)
    {
        // 三级开关:①全局 sys_http_log ②客户端级(配置 HttpRemote:{name}:EnabledLog)③接口级(特性 IgnoreLog)
        if (!_config.GetValue<bool>("HttpLog:GlobalEnabled", true)) return await base.SendAsync(request, ct);
        //var clientEnabled = _config.GetValue($"HttpLog:Clients:{request.GetHttpClientName()}:Enabled", true);
        //if (!clientEnabled || request.GetRequestMetadata<HttpApiAttr>()?.IgnoreLog == true) return await base.SendAsync(request, ct);

        {
            HttpMethod = request.Method.Method,
            RequestUrl = request.RequestUri?.ToString(),
            RequestHeaders = SerializeHeaders(request.Headers),
            StartTime = DateTime.Now,
        };
        if (request.Content is not null) log.RequestBody = await request.Content.ReadAsStringAsync(ct);

        try
        {
            var response = await base.SendAsync(request, ct);
            log.EndTime = DateTime.Now;
            log.StatusCode = response.StatusCode;
            log.ResponseHeaders = SerializeHeaders(response.Headers);
            log.IsSuccessStatusCode = response.IsSuccessStatusCode;
            log.ResponseBody = await response.Content.ReadAsStringAsync(ct);
            return response;
        }
        catch (Exception ex)
        {
            log.EndTime = DateTime.Now;
            log.IsSuccessStatusCode = false;
            log.Exception = JsonSerializer.Serialize(ex);   // 序列化后可检索的异常结构
            throw;                                          // 异常继续上抛,不影响业务
        }
        finally
        {
            log.Elapsed = sw.ElapsedMilliseconds;
            await _publisher.PublishAsync("CreateHttpLog", log);   // 事件总线异步落库,不阻塞出站
        }
    }

    private static string SerializeHeaders(HttpHeaders headers) =>
        JsonSerializer.Serialize(headers.ToDictionary(h => h.Key, h => string.Join(";", h.Value)));
```

```csharp
public interface IEventSourceStorer
{
    ValueTask WriteAsync(IEventSource source, CancellationToken ct);
    ValueTask<IEventSource> ReadAsync(CancellationToken ct);
}

// Redis 双通道实现:
public sealed class RedisEventSourceStorer : IEventSourceStorer
{
    private readonly IConnectionMultiplexer _redis;     // StackExchange.Redis
    private readonly Channel<IEventSource> _channel;    // 有界内存通道(背压)
    private readonly string _serverId;                  // 配置 Cluster:ServerId,每节点唯一

    public RedisEventSourceStorer(IConnectionMultiplexer redis, IConfiguration config, int capacity = 3000)
    {
        _redis = redis;
        _serverId = config["Cluster:ServerId"] ?? Environment.MachineName;
        _channel = Channel.CreateBounded<IEventSource>(new BoundedChannelOptions(capacity)
        {
            FullMode = BoundedChannelFullMode.Wait   // 满则等待,不丢消息
        });
        // ① single 队列(Redis List):任一节点消费一次 → 写入内存 Channel
        Task.Run(async () =>
        {
            var db = _redis.GetDatabase();
            while (true)
            {
                var item = await db.ListLeftPopAsync("adminnet_eventsource_queue:single");
                if (item is null) { await Task.Delay(10); continue; }   // 轮询(或改 BLPOP 阻塞)
                await _channel.Writer.WriteAsync(JsonSerializer.Deserialize<ChannelEventSource>(item.ToString()));
            }
        });
        // ② broadcast 流(Redis Stream + 消费组):每节点以 ServerId 建组,保证每节点都收到
        //    XREADGROUP GROUP {ServerId} {ServerId} STREAMS adminnet_eventsource_queue:broadcast >
        //    收到 → 反序列化 → _channel.Writer.WriteAsync → XACK
    }

    public async ValueTask WriteAsync(IEventSource source, CancellationToken ct)
    {
        var db = _redis.GetDatabase();
        if (source is ChannelEventSource ces && ces.EventId?.StartsWith("broadcast:") == true)
        {
            await db.StreamAddAsync("adminnet_eventsource_queue:broadcast",
                "msg", JsonSerializer.Serialize(ces));        // 广播:每节点消费组各收一次
        }
        else
        {
            await db.ListRightPushAsync("adminnet_eventsource_queue:single", JsonSerializer.Serialize(source)); // 单次:任一节点 LPOP
        }
    }

    public ValueTask<IEventSource> ReadAsync(CancellationToken ct) =>
        ValueTask.FromResult(_channel.Reader.ReadAsync(ct).AsTask());
}
```

```csharp
public class RabbitMQEventSourceStore : IEventSourceStorer, IDisposable
{
    private readonly IConnection _connection;
    private readonly IChannel _channel;
    private readonly Channel<IEventSource> _channelEventSource;  // 有界内存通道(背压)
    private readonly string _routeKey;

    public RabbitMQEventSourceStore(IConnection connection, string routeKey, int capacity = 3000)
    {
        _connection = connection;
        _routeKey = routeKey;
        _channelEventSource = Channel.CreateBounded<IEventSource>(new BoundedChannelOptions(capacity)
        {
            FullMode = BoundedChannelFullMode.Wait
        });
    }

    public async Task InitAsync()
    {
        _channel = await _connection.CreateChannelAsync();
        await _channel.QueueDeclareAsync(_routeKey, durable: true, exclusive: false, autoDelete: false, arguments: null);

        var consumer = new AsyncEventingBasicConsumer(_channel);
        consumer.ReceivedAsync += async (ch, ea) =>
        {
            var json = Encoding.UTF8.GetString(ea.Body.ToArray());
            await _channelEventSource.Writer.WriteAsync(JsonSerializer.Deserialize<ChannelEventSource>(json));
            await _channel.BasicAckAsync(ea.DeliveryTag, multiple: false);   // 手动应答
        };
        await _channel.BasicConsumeAsync(queue: _routeKey, autoAck: false, consumer: consumer);
    }

    public async ValueTask WriteAsync(IEventSource source, CancellationToken ct)
    {
        if (source is not ChannelEventSource ces)
            throw new ArgumentException("仅支持 ChannelEventSource");
        var props = new BasicProperties { ContentType = "text/plain", Persistent = true }; // 持久化
        await _channel.BasicPublishAsync("", _routeKey, false, props, Encoding.UTF8.GetBytes(JsonSerializer.Serialize(ces)));
    }

    public ValueTask<IEventSource> ReadAsync(CancellationToken ct) =>
        ValueTask.FromResult(_channelEventSource.Reader.ReadAsync(ct).AsTask());

    public void Dispose()
    {
        _channel?.Dispose();
        _connection?.Dispose();
    }
}
```

```csharp
    {
        var attr = context.Attribute;   // [EventSubscribe] 上挂的重试配置(自建特性:NumRetries/RetryTimeout/ExceptionTypes)

        // ① 执行前落审计日志(Running)——[SuppressMonitor] 标记的订阅跳过(防日志事件自身递归)
        var log = new SysLogEvent
        {
            RunId = context.Source.EventId,
            EventId = context.Source.EventId,
            Status = EventStatusEnum.Running,
            ExecutingTime = context.ExecutingTime,
        };
        if (!context.HandlerMethod.IsDefined(typeof(SuppressMonitorAttribute)))
        {
            try
            {
                log.HandlerMethod = context.HandlerMethod.Name;
                log.Payload = JsonSerializer.Serialize(context.Source.Payload);
            }
            catch (Exception ex) { _logger.LogError(ex, "序列化事件参数失败: {EventId}", context.Source.EventId); }
            await _logDb.CopyNew().Insertable(log).ExecuteCommandAsync();
        }

        // ② 重试执行(失败写 RetryRecord,重试耗尽记 Error)
        var exceptionList = new List<string>();
        var sw = Stopwatch.StartNew();
        try
        {
            await RetryHelper.InvokeAsync(async () =>
            {
                try { await handler(context); log.Status = EventStatusEnum.Success; }
                catch (Exception ex)
                {
                    log.Status = EventStatusEnum.Retry;
                    exceptionList.Add($"{DateTime.Now:yyyy-MM-dd HH:mm:ss}: {ex.Message}");
                    throw;
                }
            }, attr?.NumRetries ?? 0, attr?.RetryTimeout ?? 1000, attr?.ExceptionTypes);
        }
        finally
        {
            log.ExecutedTime = DateTime.Now;
            log.Elapsed = sw.ElapsedMilliseconds;
            log.RetryRecord = string.Join("\n", exceptionList);
            if (exceptionList.Count >= (attr?.NumRetries ?? 0) && exceptionList.Count > 0) log.Status = EventStatusEnum.Error;
            if (log.Status != EventStatusEnum.Running) await _logDb.CopyNew().Updateable(log).ExecuteCommandAsync();
        }
    }
}

// 重试辅助(替代 NewLife.Core Retry.InvokeAsync):
public static class RetryHelper
{
    public static async Task InvokeAsync(Func<Task> action, int numRetries, int retryTimeoutMs, Type[]? exceptionTypes = null)
    {
        for (var i = 0; i <= numRetries; i++)
        {
            try { await action(); return; }
            catch (Exception ex) when (exceptionTypes is null || exceptionTypes.Any(t => t.IsAssignableFrom(ex.GetType())))
            {
                if (i == numRetries) throw;
                await Task.Delay(retryTimeoutMs);
            }
        }
    }
}
```

```csharp
    public static async Task<List<T>> ReliableTake<T>(string topic, string group, string consumer, int count)

        var entries = await db.StreamReadGroupAsync($"myQueue:{topic}", group, consumer, 0, count);
        return entries.Select(e => JsonSerializer.Deserialize<T>((string)e["msg"]!)).ToList();

    public static Task AckReliable(string topic, string group, EntryId id) =>
        _muxer!.GetDatabase().StreamAcknowledgeAsync($"myQueue:{topic}", group, id);

    // ③ 延迟队列(ZSet):score=到期时间戳;后台 BackgroundService 轮询 ZRANGEBYSCORE 0 now 取到期 → 移入 List
    public static Task AddDelay<T>(string topic, T value, int delaySeconds)
    {
        var db = _muxer!.GetDatabase();
        var key = $"myDelay:{topic}";
        return db.SortedSetAddAsync(key, JsonSerializer.Serialize(value), DateTimeOffset.UtcNow.AddSeconds(delaySeconds).ToUnixTimeSeconds());
    }

    // ④ 分布式锁(替代 BeginCacheLock):SET NX PX + Lua 校验释放

    {

        return ok ? token : null;

    public static async Task ReleaseCacheLockAsync(string key, string token)

        const string lua = "if redis.call('get', KEYS[1]) == ARGV[1] then return redis.call('del', KEYS[1]) else return 0 end";
```

```csharp
    private readonly IOptions<RabbitMqOptions> _options;
    private readonly IMessageHandler _handler;

    public RabbitMqConsumer(IOptions<RabbitMqOptions> options, IMessageHandler handler, ILoggerFactory loggerFactory)

            HostName = _options.Value.HostName,

            VirtualHost = _options.Value.VirtualHost

                await _handler.HandleMessageAsync(message);

                await channel.BasicNackAsync(deliveryTag: ea.DeliveryTag, multiple: false, requeue: true); // 失败回队
            }
        };
        await channel.BasicConsumeAsync(queue: _handler.QueueName, autoAck: false, consumer: consumer);
    }
```

```csharp
                    // 防重入:Concurrent=false 时先取分布式锁/running 标记(见 5.16 JobMonitor)
                    using var scope = _root.CreateScope();

                    var record = new SysJobTriggerRecord { JobId = jobId, TriggerId = trigger.TriggerId, LastRunTime = now };
                    try
                    {
                        await jobInstance.ExecuteAsync(new JobExecutingContext(jobId, trigger), stoppingToken);
                        record.Status = 1;
                    }
                    catch (Exception ex)
                    {
                        record.Status = 2;
                        record.Exception = ex.Message;
                    }
                    finally
                    {
                        await scope.ServiceProvider.GetRequiredService<ISqlSugarClient>()
                            .CopyNew().Insertable(record).ExecuteCommandAsync();   // 运行留痕
                    }
                }
            }
            await Task.Delay(5000, stoppingToken);
        }
    }
}
```

```csharp
// ① 单节点调度选举(BackgroundService,替代 IJobClusterServer):
public class JobClusterElectionService : BackgroundService
{
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly IConfiguration _config;
    private readonly Random _random = new();
    private string ClusterId => _config["Cluster:ServerId"] ?? Environment.MachineName;

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        // 启动登记:upsert SysJobCluster{ClusterId, Status=Waiting}
        using (var scope = _scopeFactory.CreateScope())
        {
            var db = scope.ServiceProvider.GetRequiredService<ISqlSugarClient>().CopyNew();
            if (await db.Queryable<SysJobCluster>().AnyAsync(u => u.ClusterId == ClusterId))
                await db.Updateable<SysJobCluster>().SetColumns(u => u.Status == ClusterStatus.Waiting)
                    .Where(u => u.ClusterId == ClusterId).ExecuteCommandAsync();
            else
                await db.Insertable(new SysJobCluster { ClusterId = ClusterId, Status = ClusterStatus.Waiting }).ExecuteCommandAsync();
        }

        // 选举循环:随机 3~4s 心跳(错开集群同时启动),取锁后判定是否唯一 Working 节点
        while (!stoppingToken.IsCancellationRequested)
        {
            await Task.Delay(3000 + _random.Next(500, 1000), stoppingToken);
            var token = await RedisQueue.BeginCacheLockAsync("lock:JobClusterServer:Election", msTimeout: 1000);
            if (token is null) continue;
            try
            {
                using var scope = _scopeFactory.CreateScope();
                var db = scope.ServiceProvider.GetRequiredService<ISqlSugarClient>().CopyNew();
                if (await db.Queryable<SysJobCluster>().AnyAsync(u => u.Status == ClusterStatus.Working)) continue;
                // 无 Working 节点(全部宕机/首次启动)→ 本节点接管
                await db.Updateable(new SysJobCluster { Status = ClusterStatus.Working })
                    .Where(u => u.ClusterId == ClusterId).ExecuteCommandAsync();
                break;   // 接管成功:本节点启动 JobSchedulerService,其余节点待机
            }
            finally
            {
                await RedisQueue.ReleaseCacheLockAsync("lock:JobClusterServer:Election", token);
            }
        }

        // 正常停机 → 标记 Crashed,触发其他节点重选举
        using (var scope = _scopeFactory.CreateScope())
        {
            var db = scope.ServiceProvider.GetRequiredService<ISqlSugarClient>().CopyNew();
            await db.Updateable(new SysJobCluster { Status = ClusterStatus.Crashed })
                .Where(u => u.ClusterId == ClusterId).ExecuteCommandAsync();
        }
    }
}

// ② 执行监视(替代 IJobMonitor):防重入 + 失败通知
public class JobMonitor
{
    // OnExecuting:取分布式锁 {dis:lock:job:{jobId}}(抢不到=别处正在跑)→ 写 {key}:running 标记(60min 过期)
    public async Task<bool> GuardAsync(string jobId)
    {
        var token = await RedisQueue.BeginCacheLockAsync($"dis:lock:job:{jobId}", throwOnFailure: false);
        if (token is null) return false;
        // 释放 token 的存活交给调度器 finally 统一释放;另写 running 标记供界面展示"执行中"
        return true;
    }
    // OnExecuted:删 running 标记;异常 → ILogger + 邮件事件 + SysLogEx(ControllerName="定时任务")
}
```

```csharp
// 中间件(替代 LogExceptionHandler + IGlobalExceptionHandler):
public class ExceptionMiddleware
{
    private readonly RequestDelegate _next;
    private readonly ILogger<ExceptionMiddleware> _logger;
    private readonly IEventPublisher _publisher;

    public ExceptionMiddleware(RequestDelegate next, ILogger<ExceptionMiddleware> logger, IEventPublisher publisher)
    {
        _next = next;
        _logger = logger;
        _publisher = publisher;
    }

    public async Task InvokeAsync(HttpContext context)
    {
        try
        {
            await _next(context);
        }
        catch (Exception ex)
        {
            // 业务异常(AppException 及其子类):已知错误,400 系;未捕获:500,不泄露堆栈
            var (statusCode, message) = ex is AppException app
                ? (app.StatusCode, app.Message)
                : (StatusCodes.Status500InternalServerError, "系统繁忙,请稍后重试");

            _logger.LogError(ex, "全局异常: {Url} {Message}", context.Request.Path, ex.Message);   // 堆栈只进日志

            // 落库与邮件走事件总线异步(与 baseadmin 同构):处理链路不直接写库
            var sysLogEx = new SysLogEx
            {
                Account = context.User.FindFirst(JwtClaimNames.Account)?.Value,
                ControllerName = (context.GetEndpoint()?.Metadata.GetMetadata<ControllerActionDescriptor>()?.ControllerName),
                ActionName = context.GetEndpoint()?.Metadata.GetMetadata<ControllerActionDescriptor>()?.ActionName,
                Message = $"异常信息:{ex.Message}\n异常来源:{ex.Source}\n堆栈信息:{ex.StackTrace}",
                HttpMethod = context.Request.Method,
                RequestUrl = $"{context.Request.Scheme}://{context.Request.Host}{context.Request.Path}",
                RemoteIp = context.Connection.RemoteIpAddress?.ToString(),
                Browser = context.Request.Headers["User-Agent"].ToString(),
                TraceId = context.TraceStart ?? context.TraceIdentifier,      // 替代 App.GetTraceId
                ThreadId = Environment.CurrentManagedThreadId,                // 替代 App.GetThreadId
                LogDateTime = DateTime.Now,
                LogLevel = LogLevel.Error
            };
            await _publisher.PublishAsync("CreateExLog", sysLogEx);
            await _publisher.PublishAsync("SendErrorMail", ex);

            await context.Response.WriteAsJsonAsync(ApiResult.Fail(statusCode, message), context.RequestAborted);
        }
    }
}
```

```csharp
        _identityService = identityService;
        _db = db;
    }

    // 统一查询入口(baseadmin POST /apijson/get):
    // 入参 JSON 语法:{"SYS_USER[]": { "@order": "Id desc", "query": { "Status": 1 } }, "page": 1, "count": 10, "total@": ""}
    [HttpPost("get")]
    public ApiResult<object> Query([FromBody] JObject body)
    {
        // @database 指定多库:body["@database"] → _db.AsTenant().GetConnectionScope(database)
        return ApiResult.Ok(_selectTable.Query(body));
    }

    // 新增(baseadmin POST /apijson/add):逐表权限检查 → 单条/批量插入
    [HttpPost("add")]
    public async Task<ApiResult<object>> Add([FromBody] JObject tables)
    {
        // baseadmin [UnitOfWork] 特性 → 显式事务
        await _db.Ado.BeginTranAsync();
        try
        {
            var ht = new JObject();
            foreach (var prop in tables.Properties())
            {
                var tableName = prop.Name;
                var role = _identityService.GetRole(); // Claim 角色名 → APIJSONOptions.Roles
                if (!role.Insert.Table.Contains(tableName, StringComparer.CurrentCultureIgnoreCase))
                    throw new AppException($"没权限添加{tableName}");

                if (prop.Value is JArray arr) // 批量插入
                {
                    var ids = arr.Select(r => _selectTable.InsertSingle(tableName, r.ToObject<JObject>()!, role)).ToList();
                    ht[tableName] = JToken.FromObject(new { id = ids, count = ids.Count });
                }
                else // 单条插入
                {
                    var id = _selectTable.InsertSingle(tableName, prop.Value.ToObject<JObject>()!, role);
                    ht[tableName] = JToken.FromObject(new { id });
                }
            }
            await _db.Ado.CommitTranAsync();
            return ApiResult.Ok(ht);
        }
        catch
        {
            await _db.Ado.RollbackTranAsync();
            throw;
        }
    }
}

// 装配:IConfigurableOptions → IOptions 标准绑定(节名=类名,沿用 baseadmin 约定)

    services.AddSingleton<SelectTable>();   // 无状态解析器(依赖 ISqlSugarClient,实际按 SqlSugar 生命周期注册)
    services.AddScoped<IdentityService>();  // 依赖 IHttpContextAccessor 取 Claim 角色
```

```csharp
            // 其他库(达梦):Insertable/Updateable 逐条(达梦 BulkMerge 不支持,400 条以上溢出)
            await _db.Ado.CommitTranAsync();
        }
        catch
        {
            await _db.Ado.RollbackTranAsync();
            throw;
        }
    }
}
```

### 核心代码参考

```csharp
// baseadmin:SysOpenAccessService(SysOpenAccess 表 + HMAC-SHA256 签名生成/校验)
//            + Furion 内置 AddSignatureAuthentication(SignatureAuthenticationEvent:
//              OnGetAccessSecret 按 AccessKey 取密钥 / OnValidated 注入 Claim)
// 零 Furion 目标:自定义 AuthenticationHandler(scheme 名 "Signature")

public class SignatureAuthenticationHandler : AuthenticationHandler<AuthenticationOptions>
{
    private readonly SysOpenAccessService _openAccess;
    private readonly IConnectionMultiplexer _redis;

    protected override async Task<AuthenticateResult> HandleAuthenticateAsync()
    {
        // 1) 请求头:AccessKey / Timestamp / Nonce / Signature
        if (!Request.Headers.TryGetValue("AccessKey", out var ak))
            return AuthenticateResult.NoResult();
        var accessKey = ak.ToString();

        // 2) 查身份(baseadmin GetByKey:缓存 KeyOpenAccess+accessKey → SysOpenAccess 含 BindUser/SysOrg)
        var openAccess = await _openAccess.GetByKey(accessKey);
        if (openAccess == null) return AuthenticateResult.Fail("开放接口身份不存在");

        // 3) 防重放:Timestamp 偏差窗口(如 ±5 分钟)+ Nonce 一次性(Redis SetNX,TTL=窗口)
        var timestamp = long.Parse(Request.Headers["Timestamp"].ToString());
        var nonce = Request.Headers["Nonce"].ToString();
        if (Math.Abs(DateTimeOffset.UtcNow.ToUnixTimeMilliseconds() - timestamp) > 5 * 60 * 1000)
            return AuthenticateResult.Fail("时间戳过期");
        if (!await _redis.GetDatabase().StringSetAsync(
                $"sig:nonce:{nonce}", "1", TimeSpan.FromMinutes(5), When.NotExists))
            return AuthenticateResult.Fail("Nonce 已使用");

        // 4) 重算签名:拼接顺序 METHOD&URL&AccessKey&Timestamp&Nonce → HMAC-SHA256(AccessSecret) → Base64
        var parameter = $"{Request.Method.ToUpper()}&{Request.Path}&{accessKey}&{timestamp}&{nonce}";
        using var hmac = new HMACSHA256(Encoding.UTF8.GetBytes(openAccess.AccessSecret));
        var sign = Convert.ToBase64String(hmac.ComputeHash(Encoding.UTF8.GetBytes(parameter)));
        if (sign != Request.Headers["Signature"].ToString())
            return AuthenticateResult.Fail("签名校验失败");

        // 5) Claim 注入(baseadmin OnValidated 原样:UserId/TenantId/Account/RealName/AccountType/OrgId/OrgName/OrgType/TokenVersion)
        var identity = new ClaimsIdentity(
        [
            new(ClaimConst.UserId, openAccess.BindUserId + ""),
            new(ClaimConst.TenantId, openAccess.BindTenantId + ""),
            new(ClaimConst.Account, openAccess.BindUser.Account + ""),
            new(ClaimConst.RealName, openAccess.BindUser.RealName),
            new(ClaimConst.AccountType, ((int)openAccess.BindUser.AccountType).ToString()),
            new(ClaimConst.OrgId, openAccess.BindUser.OrgId + ""),
            new(ClaimConst.OrgName, openAccess.BindUser.SysOrg?.Name + ""),
            new(ClaimConst.TokenVersion, openAccess.BindUser.TokenVersion + ""),
        ], Scheme.Name);
        return AuthenticateResult.Success(new AuthenticationTicket(new ClaimsPrincipal(identity), Scheme.Name));
    }
}

// Program.cs 注册(与 JwtBearer 并列):
services.AddAuthentication()
    .AddJwtBearer(/* 见 6.4 */)
    .AddScheme<AuthenticationOptions, SignatureAuthenticationHandler>("Signature", _ => { });
// 开放接口 Controller 标注:[Authorize(AuthenticationSchemes = "Signature")]

// 签名生成接口(SysOpenAccessService.GenerateSignature,baseadmin 原样):
var parameter = $"{input.Method.ToUpper()}&{input.Url}&{input.AccessKey}&{input.Timestamp}&{input.Nonce}";
using var hmac = new HMACSHA256(Encoding.UTF8.GetBytes(input.AccessSecret));
var signature = Convert.ToBase64String(hmac.ComputeHash(Encoding.UTF8.GetBytes(parameter)));
```

### Furion 依赖替换点

| baseadmin(Furion) | 原生 |
|---|---|
| `AddSignatureAuthentication` + `SignatureAuthenticationEvent`(Furion 内置 Signature 认证 scheme:请求头解析/取密钥/校验/Claim 注入全由框架事件驱动) | 自定义 `AuthenticationHandler<AuthenticationOptions>`(HMAC-SHA256 校验 + Claim 注入,逻辑见上) |
| `IDynamicApiController` + `ITransient`(SysOpenAccessService) | 原生 Controller 路由 + 显式 DI |
| `Oops.Oh` | `AppException` + 全局异常中间件 |

### 开发注意

- 身份更新/删除必须删缓存(KeyOpenAccess+accessKey),否则密钥变更不生效(baseadmin 已做,移植保留)
- Nonce 一次性存储(Redis SetNX + TTL)配合 Timestamp 窗口防重放;窗口两端(生成方/校验方)必须一致
- 签名拼接顺序 `METHOD&URL&AccessKey&Timestamp&Nonce` 与生成接口保持完全一致(大小写/URL 形态是常见踩坑点)
